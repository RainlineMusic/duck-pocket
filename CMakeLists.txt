cmake_minimum_required(VERSION 3.22)
project(DuckPocket VERSION 1.0.1)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED JUCE_DIR)
 set(JUCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../JUCE")
endif()
set(DUCK_SANITIZER "none" CACHE STRING "Debug instrumentation: none, address, undefined, thread")
set_property(CACHE DUCK_SANITIZER PROPERTY STRINGS none address undefined thread)
if(NOT DUCK_SANITIZER STREQUAL "none")
 if(MSVC)
  if(NOT DUCK_SANITIZER STREQUAL "address")
   message(FATAL_ERROR "MSVC supports the address configuration here")
  endif()
  add_compile_options(/fsanitize=address)
 else()
  if(DUCK_SANITIZER STREQUAL "address")
   set(DUCK_SANITIZER_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(DUCK_SANITIZER STREQUAL "undefined")
   set(DUCK_SANITIZER_FLAGS -fsanitize=undefined -fno-omit-frame-pointer)
  elseif(DUCK_SANITIZER STREQUAL "thread")
   set(DUCK_SANITIZER_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
  else()
   message(FATAL_ERROR "Unknown DUCK_SANITIZER")
  endif()
  add_compile_options(${DUCK_SANITIZER_FLAGS})
  add_link_options(${DUCK_SANITIZER_FLAGS})
 endif()
endif()
if(WIN32)
 include(cmake/ApplyWindowsVBlankFix.cmake)
 duck_apply_windows_vblank_fix("${JUCE_DIR}")
endif()
add_subdirectory(${JUCE_DIR} JUCE)
set(DUCK_DEFAULT_FORMATS VST3 AAX Standalone)
if(APPLE)
 list(APPEND DUCK_DEFAULT_FORMATS AU)
endif()
set(DUCK_FORMATS "${DUCK_DEFAULT_FORMATS}" CACHE STRING "Plugin formats")
option(DUCK_ENABLE_CLAP "Build CLAP using clap-juce-extensions" ON)
if(DUCK_ENABLE_CLAP)
 include(FetchContent)
 FetchContent_Declare(clap_juce_extensions
  GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git
  GIT_TAG 7adee3a1bd4684d4caa5601100e364abccff4b4d
  GIT_SUBMODULES_RECURSE TRUE)
 FetchContent_MakeAvailable(clap_juce_extensions)
endif()
juce_add_plugin(DuckPocket COMPANY_NAME "Rainline Music" COMPANY_WEBSITE "https://github.com/RainlineMusic/phase-pocket" COMPANY_EMAIL "rainlinemusic@gmail.com" BUNDLE_ID "ru.rainlinemusic.duckpocket" AAX_IDENTIFIER "ru.rainlinemusic.duckpocket" PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE DkPk FORMATS ${DUCK_FORMATS} PRODUCT_NAME "Duck Pocket" DESCRIPTION "Lookahead dynamic sidechain ducker" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics AAX_CATEGORY Dynamics)
juce_generate_juce_header(DuckPocket)
target_sources(DuckPocket PRIVATE Source/SidechainFilter.h Source/PocketDSP.h Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_compile_definitions(DuckPocket PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
if(MSVC)
 target_compile_options(DuckPocket PRIVATE /utf-8)
else()
 target_compile_options(DuckPocket PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
target_link_libraries(DuckPocket PRIVATE juce::juce_cryptography juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
add_executable(PocketV10Test Tests/v10_test.cpp)
add_executable(PocketShortDurationTest Tests/short_duration_test.cpp)
if(MSVC)
 target_compile_options(PocketV10Test PRIVATE /utf-8)
 target_compile_options(PocketShortDurationTest PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME PocketV10 COMMAND PocketV10Test)
add_test(NAME PocketShortDuration COMMAND PocketShortDurationTest)
option(DUCK_ENABLE_OPENGL "Enable editor OpenGL renderer (runtime fallback retained)" ON)
option(DUCK_BUILD_UI_TESTS "Build processor/UI integration tests" OFF)
if(DUCK_ENABLE_OPENGL)
 target_link_libraries(DuckPocket PRIVATE juce::juce_opengl)
endif()
target_compile_definitions(DuckPocket PUBLIC DUCK_ENABLE_OPENGL=$<BOOL:${DUCK_ENABLE_OPENGL}>)
foreach(test_name numerical relative_duration null mix_attack lookahead)
 add_executable(Pocket_${test_name} Tests/${test_name}_test.cpp)
 add_test(NAME Pocket_${test_name} COMMAND Pocket_${test_name})
endforeach()
juce_add_binary_data(DuckAssets SOURCES Assets/Fonts/Inter-Regular.ttf Assets/Fonts/Inter-Medium.ttf Assets/Designer/DialLarge.png Assets/Designer/DialSmall.png Assets/Designer/TitleInfluence.svg Assets/Designer/TitleDuration.svg Assets/Designer/TitleOutput.svg Assets/Designer/TitleMS.svg Assets/Designer/Logo.svg Assets/Audio/Duck44100.wav Assets/Audio/Duck48000.wav Assets/Audio/Duck88200.wav Assets/Audio/Duck96000.wav Assets/Audio/Duck176400.wav Assets/Audio/Duck192000.wav)
set_target_properties(DuckAssets PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
target_link_libraries(DuckPocket PUBLIC DuckAssets)
if(DUCK_BUILD_UI_TESTS)
 juce_add_console_app(PocketProcessorTest PRODUCT_NAME "Duck processor tests")
 juce_generate_juce_header(PocketProcessorTest)
 target_sources(PocketProcessorTest PRIVATE Tests/processor_test.cpp)
 target_include_directories(PocketProcessorTest PRIVATE Source ${CMAKE_CURRENT_BINARY_DIR}/DuckPocket_artefacts/JuceLibraryCode)
 target_link_libraries(PocketProcessorTest PRIVATE DuckPocket juce::juce_audio_utils juce::juce_cryptography)
 add_test(NAME PocketProcessor COMMAND PocketProcessorTest)
endif()
if(DUCK_BUILD_UI_TESTS)
 juce_add_console_app(PocketUITest PRODUCT_NAME "Duck UI tests")
 juce_generate_juce_header(PocketUITest)
 target_sources(PocketUITest PRIVATE Tests/ui_test.cpp)
 target_include_directories(PocketUITest PRIVATE Source ${CMAKE_CURRENT_BINARY_DIR}/DuckPocket_artefacts/JuceLibraryCode)
 target_compile_definitions(PocketUITest PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(PocketUITest PRIVATE DuckPocket juce::juce_audio_utils juce::juce_cryptography)
 if(DUCK_ENABLE_OPENGL)
  target_link_libraries(PocketProcessorTest PRIVATE juce::juce_opengl)
  target_link_libraries(PocketUITest PRIVATE juce::juce_opengl)
 endif()
endif()
# Carry the embedded-font license in every delivered plug-in bundle, before
# CI signs/notarises it. Fonts remain embedded; only the license is copied.
foreach(format VST3 AAX AU)
 if(TARGET DuckPocket_${format})
  add_custom_command(TARGET DuckPocket_${format} POST_BUILD
   COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:DuckPocket_${format}>/../Resources"
   COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "$<TARGET_FILE_DIR:DuckPocket_${format}>/../Resources/Inter-OFL.txt"
   VERBATIM)
 endif()
endforeach()

 endif()

  COMMAND ${CMAKE_COMMAND} -E make_directory "${DUCK_CLAP_LICENSE_DIR}"

  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "${DUCK_CLAP_LICENSE_DIR}/Inter-OFL.txt"
  VERBATIM)
endif()
